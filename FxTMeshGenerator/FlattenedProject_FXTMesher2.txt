    5:     /// <summary>
    6:     /// Mathematical utility methods for mesh generation.
    7:     /// </summary>
    8:     public static class MathHelper
    9:     {
   10:         private const double OverlapTolerance = 1e-5;
   11: 
   12:         /// <summary>
   13:         /// Creates a 2D vector from point p1 to point p2.
   14:         /// </summary>
   15:         public static Point2D MakeVector2D(Point2D p1, Point2D p2)
   16:         {
   17:             return new Point2D(p2.X - p1.X, p2.Y - p1.Y);
   18:         }
   19: 
   20:         /// <summary>
   21:         /// Calculates the angle between two vectors, measured counter-clockwise from the positive x-axis.
   22:         /// </summary>
   23:         public static double CalculateAngleBetweenVectors(Point2D vec1, Point2D vec2)
   24:         {
   25:             double dotProduct = vec1.X * vec2.X + vec1.Y * vec2.Y;
   26:             double norm1 = Math.Sqrt(vec1.X * vec1.X + vec1.Y * vec1.Y);
   27:             double norm2 = Math.Sqrt(vec2.X * vec2.X + vec2.Y * vec2.Y);
   28: 
   29:             double theta = Math.Acos(dotProduct / (norm1 * norm2));
   30: 
   31:             // Check for vectors pointing downward (negative y component)
   32:             Point2D unit = new Point2D(1, 0);
   33:             bool vec1IsUnit = Math.Abs(vec1.X - 1.0) < 1e-10 && Math.Abs(vec1.Y) < 1e-10;
   34:             bool vec2IsUnit = Math.Abs(vec2.X - 1.0) < 1e-10 && Math.Abs(vec2.Y) < 1e-10;
   35: 
   36:             if ((vec1IsUnit && vec2.Y < 0) || (vec2IsUnit && vec1.Y < 0))
   37:             {
   38:                 theta = 2.0 * Math.PI - theta;
   39:             }
   40: 
   41:             return theta;
   42:         }
   43: 
   44:         /// <summary>
   45:         /// Calculates Euclidean distance between two points.
   46:         /// </summary>
   47:         public static double CalcDistanceBetweenTwoPoints(Point2D p1, Point2D p2)
   48:         {
   49:             double dx = p2.X - p1.X;
   50:             double dy = p2.Y - p1.Y;
   51:             return Math.Sqrt(dx * dx + dy * dy);
   52:         }
   53: 
   54:         /// <summary>
   55:         /// Checks if two points overlap within tolerance.
   56:         /// </summary>
   57:         public static bool OverlapCheck(Point2D p1, Point2D p2)
   58:         {
   59:             return CalcDistanceBetweenTwoPoints(p1, p2) <= OverlapTolerance;
   60:         }
   61:     }
   62: }

  102: 
  103:             using var sw = new StreamWriter(path + ".vtk");
  104:             sw.WriteLine("# vtk DataFile Version 3.0");
  105:             sw.WriteLine("FxT Mesh output");
  106:             sw.WriteLine("ASCII");
  107:             sw.WriteLine("DATASET UNSTRUCTURED_GRID");

  118:             foreach (var elem in mesh.Elements)
  119:             {

  121:             }

  134:                     // Remapping: [2,4,0,3,5,1] -> [0,1,2,3,4,5]

  136:                     for (int i = 0; i < 6; i++)

  138:                         int idx = FindNodeIndex(mesh.GlobalNodes, elem.Nodes[reorderMap[i]]);
  139:                         sw.Write($" {idx}");
  140:                     }
  141:                 }

  143:                 else if (elem is TriangleElement tri6 && tri6.NodeCount == 6 && tri6.Phase == ElementPhase.Matrix)
  144:                 {
  145:                     // Our node order: [0,1,2]=corners, [3]=mid12, [4]=mid01, [5]=mid20
  146:                     // VTK expects: [0,1,2]=corners, [3]=mid01, [4]=mid12, [5]=mid20

  150:                     {
  151:                         int idx = FindNodeIndex(mesh.GlobalNodes, elem.Nodes[reorderMap[i]]);
  152:                         sw.Write($" {idx}");
  153:                     }

  161:                     //   |     |
  162:                     //   7     3
  163:                     //   |     |
  164:                     //   0--1--2
  165:                     //
  166:                     // VTK expects: [0,1,2,3,4,5,6,7] = corner, corner, corner, corner, mid, mid, mid, mid

  168:                     //   3--6--2
  169:                     //   |     |
  170:                     //   7     5
  171:                     //   |     |
  172:                     //   0--4--1
  173:                     //
  174:                     // Remapping: [0,2,4,6,1,3,5,7] -> [0,1,2,3,4,5,6,7]
  175:                     int[] reorderMap = new int[] { 0, 2, 4, 6, 1, 3, 5, 7 };
  176:                     for (int i = 0; i < 8; i++)
  177:                     {
  178:                         int idx = FindNodeIndex(mesh.GlobalNodes, elem.Nodes[reorderMap[i]]);
  179:                         sw.Write($" {idx}");
  180:                     }
  181:                 }

  184:                     // Standard order for all other elements

  187:                         int idx = FindNodeIndex(mesh.GlobalNodes, elem.Nodes[i]);

  208:             sw.WriteLine("SCALARS element_phase int 1");
  209:             sw.WriteLine("LOOKUP_TABLE default");
  210:             foreach (var elem in mesh.Elements)
  211:             {
  212:                 sw.WriteLine(((int)elem.Phase).ToString(CultureInfo.InvariantCulture));
  213:             }
  214:

  218:             foreach (var elem in mesh.Elements)

  220:                 sw.WriteLine(elem.Id.ToString(CultureInfo.InvariantCulture));

  322:         {

  324:             var processedPairs = new HashSet<(int, int)>();

  336:                     new[] { tri1[2], tri1[0] }
  337:                 };

  339:                 foreach (var edge in edges)
  340:                 {
  341:                     // Find adjacent triangle sharing this edge
  342:                     for (int j = i + 1; j < triangleCount; j++)

  344:                         if (processedPairs.Contains((i, j)))

  346: 
  347:                         var tri2 = GetTriangleNodes(j, triangles);
  348: 
  349: 
  350:                         // Check if tri2 shares this edge
  351:                         int matchCount = 0;
  352:                         foreach (var nodeIdx in tri2)

  362:                             // If concave, swapping would create overlapping triangles

  371:                             var p_shared1 = nodes[edge[0]].P;
  372:                             var p_shared2 = nodes[edge[1]].P;
  373: 
  374:                             // COMPLETE Convexity check for quadrilateral:
  375:                             // 1. Unique nodes must be on OPPOSITE sides of current shared edge

  470:         /// Ensures proper winding order is maintained using geometric orientation.
  471:         /// </summary>
  472:         private void PerformEdgeSwap(int tri1Idx, int tri2Idx, int[] sharedEdge, int[] triangles, List<Node> nodes)
  473:         {
  474:             // Get current triangles
  475:             var tri1 = GetTriangleNodes(tri1Idx, triangles);
  476:             var tri2 = GetTriangleNodes(tri2Idx, triangles);
  477: 
  478:             // CRITICAL: Recompute the actual shared edge from the current triangle state
  479:             // The passed-in sharedEdge might be stale if triangles were modified
  480:             var actualSharedNodes = tri1.Intersect(tri2).ToArray();
  481: 
  482:             if (actualSharedNodes.Length != 2)
  483:             {
  484:                 // Invalid configuration - triangles don't share exactly one edge
  485:                 return;
  486:             }
  487: 
  488:             // Use the actual shared edge, not the passed-in one
  489:             int sharedNode1 = actualSharedNodes[0];
  490:             int sharedNode2 = actualSharedNodes[1];
  491: 
  492:             // Find the two nodes that are NOT on the shared edge (these form the new diagonal)
  493:             var uniqueToTri1 = tri1.Except(actualSharedNodes).First();
  494:             var uniqueToTri2 = tri2.Except(actualSharedNodes).First();
  495: 
  496:             // Get node positions for orientation checks
  497:             var p_unique1 = nodes[uniqueToTri1].P;
  498:             var p_unique2 = nodes[uniqueToTri2].P;
  499:             var p_shared1 = nodes[sharedNode1].P;
  500:             var p_shared2 = nodes[sharedNode2].P;
  501: 
  502:             // COMPLETE CONVEXITY CHECK: The quadrilateral must be convex for the swap to be valid
  503:             // Check 1: If both unique nodes are on the SAME side of the shared edge, 
  504:             // the quad is concave and swapping would create overlapping triangles
  505:             double side1 = Side(p_shared1, p_shared2, p_unique1);
  506:             double side2 = Side(p_shared1, p_shared2, p_unique2);
  507: 
  508:             if (side1 * side2 > 0)
  509:             {
  510:                 // Both unique nodes on same side = concave quadrilateral = swap would overlap
  511:                 // Abort the swap
  512:                 return;
  513:             }
  514: 
  515:             // Check 2: Shared edge nodes must be on OPPOSITE sides of the new diagonal
  516:             // If they're on the same side, the new diagonal exits the quadrilateral
  517:             double side3 = Side(p_unique1, p_unique2, p_shared1);
  518:             double side4 = Side(p_unique1, p_unique2, p_shared2);
  519: 
  520:             if (side3 * side4 > 0)
  521:             {
  522:                 // Both shared nodes on same side of new diagonal = new diagonal exits quad
  523:                 // Swapping would create overlapping triangles
  524:                 return;
  525:             }
  526: 
  527:             int baseTri1 = tri1Idx * 3;
  528:             int baseTri2 = tri2Idx * 3;
  529: 
  530:             // Triangle 1: uniqueToTri1, sharedNode1, uniqueToTri2
  531:             // Check orientation: uniqueToTri1 -> sharedNode1 -> uniqueToTri2
  532:             double orient1 = Side(p_unique1, p_shared1, p_unique2);
  533:             if (orient1 > 0)
  534:             {
  535:                 // CCW
  536:                 triangles[baseTri1] = uniqueToTri1;
  537:                 triangles[baseTri1 + 1] = sharedNode1;
  538:                 triangles[baseTri1 + 2] = uniqueToTri2;
  539:             }
  540:             else
  541:             {
  542:                 // CW: reverse to CCW
  543:                 triangles[baseTri1] = uniqueToTri1;
  544:                 triangles[baseTri1 + 1] = uniqueToTri2;
  545:                 triangles[baseTri1 + 2] = sharedNode1;
  546:             }
  547: 
  548:             // Triangle 2: uniqueToTri2, sharedNode2, uniqueToTri1
  549:             // Check orientation: uniqueToTri2 -> sharedNode2 -> uniqueToTri1
  550:             double orient2 = Side(p_unique2, p_shared2, p_unique1);
  551:             if (orient2 > 0)
  552:             {
  553:                 // CCW
  554:                 triangles[baseTri2] = uniqueToTri2;
  555:                 triangles[baseTri2 + 1] = sharedNode2;
  556:                 triangles[baseTri2 + 2] = uniqueToTri1;
  557:             }
  558:             else
  559:             {
  560:                 // CW: reverse to CCW
  561:                 triangles[baseTri2] = uniqueToTri2;
  562:                 triangles[baseTri2 + 1] = uniqueToTri1;
  563:                 triangles[baseTri2 + 2] = sharedNode2;
  564:             }
  565:         }

  567:         /// <summary>
  568:         /// Optimizes triangulation quality by evaluating and swapping edges based on:
  569:         /// 1. Inversion prevention (CRITICAL): No fiber should cross the line connecting adjacent surface midpoints
  570:         /// 2. Quality ratio optimization (IMPORTANT): Minimize ratio of surface spacing to inversion distance
  571:         /// 
  572:         /// Uses a priority-based single-swap approach with cycle detection.
  573:         /// Operates on the full triangulation before filtering to allow optimization across periodic boundaries.
  574:         /// </summary>
  575:         private void OptimizeTriangulationQuality(List<Node> uniqueNodes, Delaunator delaunay, IReadOnlyList<Fiber> fibers, DebugOptions dOptions)
  576:         {
  577:             // Initialize log file
  578:             InitializeLogFile(dOptions);
  579: 
  580:             LogMessage($"\n=== Starting Quality-Based Triangulation Optimization ===", dOptions);
  581: 
  582:             // Configuration parameters
  583:             const double ASPECT_RATIO_THRESHOLD = 2.5; // Don't bother swapping if aspect ratio is already decent
  584:             const int MAX_ITERATIONS = 100;
  585:             const int MAX_STALL_ITERATIONS = 5; // Stop if no progress after this many iterations
  586: 
  587:             int iterationCount = 0;
  588:             int totalSwaps = 0;
  589:             int stallCount = 0;
  590:             var recentSwaps = new HashSet<(int, int)>(); // Track recent swaps to detect cycles
  591:             bool anySwapOccurred;
  592: 
  593:             do
  594:             {
  595:                 anySwapOccurred = false;
  596:                 iterationCount++;
  597: 
  598:                 // Find all interior edges
  599:                 var interiorEdges = FindAllInteriorEdges(uniqueNodes, delaunay.Triangles);
  600: 
  601:                 // Evaluate all edges and find the best single swap
  602:                 double bestPriority = -1;
  603:                 int bestTri1 = -1, bestTri2 = -1;
  604:                 int[] bestQuad = null;
  605:                 int[] bestSharedEdge = null; // Track the shared edge for the best swap
  606:                 (int inversions, int overlaps, double innerAspect, double outerAspect) bestCurrent = default;
  607:                 (int inversions, int overlaps, double innerAspect, double outerAspect) bestSwapped = default;
  608: 
  609:                 // Track statistics for this iteration
  610:                 int inversionsFound = 0;
  611:                 int qualityImprovementsFound = 0;
  612: 
  613:                 foreach (var (tri1Idx, tri2Idx, sharedEdge) in interiorEdges)
  614:                 {
  615:                     var currentQuality = EvaluateQuadrilateralQuality(
  616:                         tri1Idx, tri2Idx, uniqueNodes, delaunay.Triangles, fibers, out var quad, sharedEdge);
  617: 
  618:                     var swappedQuality = EvaluateSwappedQuadrilateralQuality(
  619:                         tri1Idx, tri2Idx, quad, sharedEdge, uniqueNodes, delaunay.Triangles, fibers);
  620: 
  621:                     // Track inversions that could be fixed
  622:                     if (swappedQuality.inversions < currentQuality.inversions)
  623:                         inversionsFound++;
  624: 
  625:                     // Track quality improvements (overlaps/aspect ratio, no inversion change)
  626:                     if (currentQuality.inversions == swappedQuality.inversions &&
  627:                         IsConfigurationBetter(swappedQuality, currentQuality))
  628:                         qualityImprovementsFound++;
  629: 
  630:                     // Check if swap is worthwhile and not a recent swap (cycle detection)
  631:                     var swapKey = (Math.Min(tri1Idx, tri2Idx), Math.Max(tri1Idx, tri2Idx));
  632:                     if (IsSwapWorthwhile(currentQuality, swappedQuality, ASPECT_RATIO_THRESHOLD) &&
  633:                         !recentSwaps.Contains(swapKey))
  634:                     {
  635:                         double priority = CalculateSwapPriority(currentQuality, swappedQuality, uniqueNodes, fibers, quad);
  636: 
  637:                         if (priority > bestPriority)
  638:                         {
  639:                             bestPriority = priority;
  640:                             bestTri1 = tri1Idx;
  641:                             bestTri2 = tri2Idx;
  642:                             bestQuad = quad;
  643:                             bestSharedEdge = sharedEdge; // Store the shared edge
  644:                             bestCurrent = currentQuality;
  645:                             bestSwapped = swappedQuality;
  646:                         }
  647:                     }
  648:                 }
  649: 
  650:                 // Perform the best swap if found
  651:                 int inversionSwapsPerformed = 0;
  652:                 int qualitySwapsPerformed = 0;
  653: 
  654:                 if (bestPriority > -1)
  655:                 {
  656:                     PerformEdgeSwap(bestTri1, bestTri2, bestSharedEdge, delaunay.Triangles, uniqueNodes);
  657:                     totalSwaps++;
  658:                     anySwapOccurred = true;
  659:                     stallCount = 0; // Reset stall counter
  660: 
  661:                     // Track this swap for cycle detection (keep last 10 swaps)
  662:                     var swapKey = (Math.Min(bestTri1, bestTri2), Math.Max(bestTri1, bestTri2));
  663:                     recentSwaps.Add(swapKey);
  664:                     if (recentSwaps.Count > 10)
  665:                     {
  666:                         recentSwaps.Remove(recentSwaps.First());
  667:                     }
  668: 
  669:                     // Categorize the swap
  670:                     if (bestSwapped.inversions < bestCurrent.inversions)
  671:                         inversionSwapsPerformed = 1;
  672:                     else if (bestSwapped.overlaps < bestCurrent.overlaps)
  673:                         inversionSwapsPerformed = 1;  // Treat overlap fixes as important
  674:                     else
  675:                         qualitySwapsPerformed = 1;
  676: 
  677:                     // Write VTK after each swap for debugging
  678:                     if (dOptions != null && dOptions.Debug)
  679:                     {
  680:                         // Convert flat triangle array to List<int[]>
  681:                         var debugTris = new List<int[]>();
  682:                         for (int t = 0; t < delaunay.Triangles.Length; t += 3)
  683:                         {
  684:                             debugTris.Add(new int[3] { delaunay.Triangles[t], delaunay.Triangles[t + 1], delaunay.Triangles[t + 2] });
  685:                         }
  686: 
  687:                         //CreateVTKFileFromTriangles(uniqueNodes,debugTris,dOptions);
  688:                         LogMessage($"  - Swapped triangles {bestTri1} and {bestTri2}: " +
  689:                                   $"inv {bestCurrent.inversions}->{bestSwapped.inversions}, " +
  690:                                   $"overlap {bestCurrent.overlaps}->{bestSwapped.overlaps}, " +
  691:                                   $"innerAR {bestCurrent.innerAspect:F2}->{bestSwapped.innerAspect:F2}", dOptions);
  692:                     }
  693:                 }
  694:                 else
  695:                 {
  696:                     stallCount++;
  697:                 }
  698: 
  699:                 // Log iteration summary
  700:                 LogMessage($"Iteration {iterationCount}: " +
  701:                           $"# Inversions={inversionsFound}, Inversions Swapped={inversionSwapsPerformed}, " +
  702:                           $"# Quality Improvements={qualityImprovementsFound}, Quality Swapped={qualitySwapsPerformed}", dOptions);
  703: 
  704:                 // Stop if stalled (no beneficial swaps found)
  705:                 if (stallCount >= MAX_STALL_ITERATIONS)
  706:                 {
  707:                     LogMessage($"Converged: No beneficial swaps found for {MAX_STALL_ITERATIONS} iterations.", dOptions);
  708:                     break;
  709:                 }
  710: 
  711:                 // Safety limit
  712:                 if (iterationCount >= MAX_ITERATIONS)
  713:                 {
  714:                     LogMessage($"Warning: Reached maximum iterations ({MAX_ITERATIONS}). Stopping optimization.", dOptions);
  715:                     break;
  716:                 }
  717:             }
  718:             while (anySwapOccurred || stallCount < MAX_STALL_ITERATIONS);
  719: 
  720:             LogMessage($"\n=== Optimization Complete ===", dOptions);
  721:             LogMessage($"Total iterations: {iterationCount}", dOptions);
  722:             LogMessage($"Total swaps performed: {totalSwaps}", dOptions);
  723: 
  724:             // Log final inversion count (edge triangles may be removed later during projection filtering)
  725:             int finalInversions = CountTotalInversions(uniqueNodes, delaunay.Triangles, fibers);
  726:             LogMessage($"Final inversion count: {finalInversions}", dOptions);
  727: 
  728:             // Close log file
  729:             CloseLogFile();
  730:         }

  732:         /// <summary>
  733:         /// Determines if a swap is worthwhile based on improvement and thresholds.
  734:         /// </summary>
  735:         private bool IsSwapWorthwhile((int inversions, int overlaps, double innerAspect, double outerAspect) current,
  736:             (int inversions, int overlaps, double innerAspect, double outerAspect) swapped, double aspectThreshold)
  737:         {
  738:             // Always swap if it reduces inversions
  739:             if (swapped.inversions < current.inversions)
  740:                 return true;
  741: 
  742:             // Don't swap if it increases inversions
  743:             if (swapped.inversions > current.inversions)
  744:                 return false;
  745: 
  746:             // Always swap if it reduces overlaps (same inversions)
  747:             if (swapped.overlaps < current.overlaps)
  748:                 return true;
  749: 
  750:             // Don't swap if it increases overlaps
  751:             if (swapped.overlaps > current.overlaps)
  752:                 return false;
  753: 
  754:             // Same inversions and overlaps: check if we should bother swapping for aspect ratio alone
  755:             // If already good (no inversions/overlaps), don't risk making it worse
  756:             if (current.inversions == 0 && current.overlaps == 0)
  757:                 return false;
  758: 
  759:             // Has problems: only swap if inner aspect ratio is bad enough and swap improves it
  760:             if (current.innerAspect > aspectThreshold && swapped.innerAspect < current.innerAspect)
  761:                 return true;
  762: 
  763:             return false;
  764:         }

  784:         /// <summary>
  785:         /// Calculates priority score for a potential edge swap.
  786:         /// Higher values indicate more important swaps.
  787:         /// Priority hierarchy:
  788:         /// 1. Inversion elimination (weight: 10000 per inversion)
  789:         /// 2. Overlap elimination (weight: 1000 per overlap)
  790:         /// 3. Inner aspect ratio improvement (weight: 100)
  791:         /// 4. Outer aspect ratio improvement (weight: 10)
  792:         /// </summary>
  793:         private double CalculateSwapPriority(
  794:             (int inversions, int overlaps, double innerAspect, double outerAspect) current,
  795:             (int inversions, int overlaps, double innerAspect, double outerAspect) swapped,
  796:             List<Node> nodes,
  797:             IReadOnlyList<Fiber> fibers,
  798:             int[] quad)
  799:         {
  800:             double priority = 0;
  801: 
  802:             // CRITICAL: Inversion reduction (weight: 10000 per inversion)
  803:             int inversionReduction = current.inversions - swapped.inversions;
  804:             if (inversionReduction > 0)
  805:             {
  806:                 priority += 10000 * inversionReduction;
  807:             }
  808: 
  809:             // VERY IMPORTANT: Overlap reduction (weight: 1000 per overlap)
  810:             int overlapReduction = current.overlaps - swapped.overlaps;
  811:             if (overlapReduction > 0)
  812:             {
  813:                 priority += 1000 * overlapReduction;
  814:             }
  815: 
  816:             // IMPORTANT: Inner aspect ratio improvement (weight: 100)
  817:             // Lower aspect ratio is better
  818:             double innerImprovement = current.innerAspect - swapped.innerAspect;
  819:             if (innerImprovement > 0)
  820:             {
  821:                 priority += 100 * innerImprovement;
  822:             }
  823: 
  824:             // NICE TO HAVE: Outer aspect ratio improvement (weight: 10)
  825:             double outerImprovement = current.outerAspect - swapped.outerAspect;
  826:             if (outerImprovement > 0)
  827:             {
  828:                 priority += 10 * outerImprovement;
  829:             }
  830: 
  831:             return priority;
  832:         }

  834:         /// <summary>
  835:         /// Evaluates the quality of a quadrilateral formed by two adjacent triangles.
  836:         /// Returns (inversionCount, overlapCount, maxInnerAspectRatio, maxOuterAspectRatio) and outputs the 4-node quadrilateral.
  837:         /// </summary>
  838:         private (int inversions, int overlaps, double innerAspect, double outerAspect) EvaluateQuadrilateralQuality(
  839:             int tri1Idx, int tri2Idx, List<Node> nodes, int[] triangles, IReadOnlyList<Fiber> fibers,
  840:             out int[] quad, int[] sharedEdge)
  841:         {
  842:             var tri1 = GetTriangleNodes(tri1Idx, triangles);
  843:             var tri2 = GetTriangleNodes(tri2Idx, triangles);
  844: 
  845:             // Find the 4 unique vertices forming the quadrilateral
  846:             var allVertices = tri1.Concat(tri2).Distinct().ToArray();
  847:             quad = allVertices;
  848: 
  849:             if (quad.Length != 4)
  850:             {
  851:                 // Degenerate case - shouldn't happen
  852:                 return (int.MaxValue, int.MaxValue, double.MaxValue, double.MaxValue);
  853:             }
  854: 
  855:             // Evaluate both triangles
  856:             int totalInversions = 0;
  857:             int totalOverlaps = 0;
  858:             double maxInnerAspect = 0;
  859:             double maxOuterAspect = 0;
  860: 
  861:             foreach (var tri in new[] { tri1, tri2 })
  862:             {
  863:                 var (inversions, overlaps, innerAspect, outerAspect) = EvaluateTriangleQuality(tri, nodes, fibers);
  864:                 totalInversions += inversions;
  865:                 totalOverlaps += overlaps;
  866:                 maxInnerAspect = Math.Max(maxInnerAspect, innerAspect);
  867:                 maxOuterAspect = Math.Max(maxOuterAspect, outerAspect);
  868:             }
  869: 
  870:             return (totalInversions, totalOverlaps, maxInnerAspect, maxOuterAspect);
  871:         }

  873:         /// <summary>
  874:         /// Evaluates what the quality would be if we swapped the diagonal of a quadrilateral.
  875:         /// Uses the EXACT same geometric orientation logic as PerformEdgeSwap to ensure consistency.
  876:         /// </summary>
  877:         private (int inversions, int overlaps, double innerAspect, double outerAspect) EvaluateSwappedQuadrilateralQuality(
  878:             int tri1Idx, int tri2Idx, int[] quad, int[] sharedEdge, List<Node> nodes, int[] triangles, IReadOnlyList<Fiber> fibers)
  879:         {
  880:             if (quad.Length != 4)
  881:                 return (int.MaxValue, int.MaxValue, double.MaxValue, double.MaxValue);
  882: 
  883:             // Get the current triangles
  884:             var tri1 = GetTriangleNodes(tri1Idx, triangles);
  885:             var tri2 = GetTriangleNodes(tri2Idx, triangles);
  886: 
  887:             // Find the two nodes that are NOT on the shared edge (these will form the new diagonal)
  888:             var uniqueToTri1 = tri1.Except(sharedEdge).First();
  889:             var uniqueToTri2 = tri2.Except(sharedEdge).First();
  890: 
  891:             int sharedNode1 = sharedEdge[0];
  892:             int sharedNode2 = sharedEdge[1];
  893: 
  894:             // Get node positions for orientation checks (EXACT SAME logic as PerformEdgeSwap)
  895:             var p_unique1 = nodes[uniqueToTri1].P;
  896:             var p_unique2 = nodes[uniqueToTri2].P;
  897:             var p_shared1 = nodes[sharedNode1].P;
  898:             var p_shared2 = nodes[sharedNode2].P;
  899: 
  900:             // Create swapped triangles with proper geometric orientation
  901:             int[] swappedTri1, swappedTri2;
  902: 
  903:             // Triangle 1: uniqueToTri1, sharedNode1, uniqueToTri2
  904:             // MUST MATCH PerformEdgeSwap line 1047
  905:             double orient1 = Side(p_unique1, p_shared1, p_unique2);
  906:             if (orient1 > 0)
  907:             {
  908:                 // CCW
  909:                 swappedTri1 = new[] { uniqueToTri1, sharedNode1, uniqueToTri2 };
  910:             }
  911:             else
  912:             {
  913:                 // CW: reverse to CCW
  914:                 swappedTri1 = new[] { uniqueToTri1, uniqueToTri2, sharedNode1 };
  915:             }
  916: 
  917:             // Triangle 2: uniqueToTri2, sharedNode2, uniqueToTri1
  918:             // MUST MATCH PerformEdgeSwap line 1065
  919:             double orient2 = Side(p_unique2, p_shared2, p_unique1);
  920:             if (orient2 > 0)
  921:             {
  922:                 // CCW
  923:                 swappedTri2 = new[] { uniqueToTri2, sharedNode2, uniqueToTri1 };
  924:             }
  925:             else
  926:             {
  927:                 // CW: reverse to CCW
  928:                 swappedTri2 = new[] { uniqueToTri2, uniqueToTri1, sharedNode2 };
  929:             }
  930: 
  931:             // Evaluate the swapped configuration
  932:             return EvaluateTwoTrianglesQuality(swappedTri1, swappedTri2, nodes, fibers);
  933:         }

  935:         /// <summary>
  936:         /// Evaluates the combined quality of two triangles.
  937:         /// Returns worst-case metrics across both triangles.
  938:         /// </summary>
  939:         private (int inversions, int overlaps, double innerAspect, double outerAspect) EvaluateTwoTrianglesQuality(
  940:             int[] tri1, int[] tri2, List<Node> nodes, IReadOnlyList<Fiber> fibers)
  941:         {

  948:         /// <summary>
  949:         /// Evaluates the quality of a single triangle using aspect ratio.
  950:         /// Returns (topologicalInversions, elementOverlapCount, innerTriangleAspectRatio, outerTriangleAspectRatio).
  951:         /// Priority for swapping (lower is better):
  952:         /// 1. Topological inversions (fiber center outside triangle edge) - CRITICAL
  953:         /// 2. Element overlaps (fiber surface outside inner triangle) - count how many fibers
  954:         /// 3. Inner triangle aspect ratio (minimize for good element quality)
  955:         /// 4. Outer triangle aspect ratio (minimize for good triangle shape)
  956:         /// Goal: minimize in order of priority.
  957:         /// </summary>
  958:         private (int topologicalInversions, int elementOverlaps, double innerAspectRatio, double outerAspectRatio) EvaluateTriangleQuality(int[] triangleNodeIndices,
  959:             List<Node> nodes, IReadOnlyList<Fiber> fibers)
  960:         {
  961:             // Get the three nodes
  962:             var nodeA = nodes[triangleNodeIndices[0]];
  963:             var nodeB = nodes[triangleNodeIndices[1]];
  964:             var nodeC = nodes[triangleNodeIndices[2]];
  965: 
  966:             // Calculate interior points for quality assessment
  967:             // For fiber nodes: use surface midpoint; for boundary nodes: use the boundary point itself
  968:             var interiorPoints = new Point2D[3];
  969:             var triangleNodes = new[] { nodeA, nodeB, nodeC };
  970: 
  971:             for (int i = 0; i < 3; i++)
  972:             {
  973:                 var currentNode = triangleNodes[i];
  974:                 var otherIndices = GetOtherIndices(i);
  975:                 var otherNode1 = triangleNodes[otherIndices[0]];
  976:                 var otherNode2 = triangleNodes[otherIndices[1]];
  977: 
  978:                 // If this is a fiber node, calculate surface point; otherwise use boundary point directly
  979:                 if (currentNode.FiberId.HasValue)
  980:                 {
  981:                     interiorPoints[i] = CalculateFiberSurfacePoint(
  982:                         currentNode.P, fibers[currentNode.FiberId.Value].Radius,
  983:                         otherNode1.P, otherNode2.P);
  984:                 }
  985:                 else
  986:                 {
  987:                     // Boundary node: use its position directly
  988:                     interiorPoints[i] = currentNode.P;
  989:                 }
  990:             }
  991: 
  992:             // Calculate quality metrics
  993:             int topologicalInversionCount = 0;
  994:             int elementOverlapCount = 0;
  995: 
  996:             for (int i = 0; i < 3; i++)
  997:             {
  998:                 // Only check fiber nodes
  999:                 if (triangleNodes[i].FiberId.HasValue)
 1000:                 {
 1001:                     var fiberCenter = triangleNodes[i].P;
 1002:                     var otherIndices = GetOtherIndices(i);
 1003:                     var interiorPoint1 = interiorPoints[otherIndices[0]];
 1004:                     var interiorPoint2 = interiorPoints[otherIndices[1]];
 1005:                     var nodePos1 = triangleNodes[otherIndices[0]].P;
 1006:                     var nodePos2 = triangleNodes[otherIndices[1]].P;
 1007: 
 1008:                     // Priority 1: Topological inversion (fiber center vs node positions)
 1009:                     // This is a triangle-level problem where fiber center crosses the edge
 1010:                     bool topologicalInversion = !SameSide(fiberCenter, interiorPoints[i], nodePos1, nodePos2);
 1011:                     if (topologicalInversion)
 1012:                         topologicalInversionCount++;
 1013: 
 1014:                     // Priority 2: Element-level overlap (fiber surface vs surface points)
 1015:                     // This determines if the interior element overlaps the fiber
 1016:                     bool elementOverlap = IsInnerTriangleInverted(fiberCenter, interiorPoints[i], interiorPoint1, interiorPoint2);
 1017:                     if (elementOverlap)
 1018:                         elementOverlapCount++;
 1019:                 }
 1020:             }
 1021:

 1023:             double innerAspectRatio = CalculateAspectRatio(interiorPoints[0], interiorPoints[1], interiorPoints[2]);

 1029:             double outerAspectRatio = CalculateAspectRatio(outerNodePos0, outerNodePos1, outerNodePos2);

 1039:         {
 1040:             // checks that the fiber center is on the same side of the line defined by surfacePoint1 and surfacePoint2 as the surface point of the fiber itself

 1044:         {
 1045:             return (p3.X - p2.X) * (q.Y - p2.Y)
 1046:                  - (p3.Y - p2.Y) * (q.X - p2.X);

 1057:         /// <summary>
 1058:         /// Calculates the aspect ratio of a triangle (longest edge / shortest edge).
 1059:         /// Lower values are better (1.0 = equilateral).
 1060:         /// </summary>
 1061:         private double CalculateAspectRatio(Point2D p1, Point2D p2, Point2D p3)
 1062:         {
 1063:             double edge1 = MathHelper.CalcDistanceBetweenTwoPoints(p1, p2);
 1064:             double edge2 = MathHelper.CalcDistanceBetweenTwoPoints(p2, p3);
 1065:             double edge3 = MathHelper.CalcDistanceBetweenTwoPoints(p3, p1);
 1066: 
 1067:             double maxEdge = Math.Max(edge1, Math.Max(edge2, edge3));
 1068:             double minEdge = Math.Min(edge1, Math.Min(edge2, edge3));
 1069: 
 1070:             if (minEdge < 1e-10)
 1071:                 return double.MaxValue; // Degenerate triangle
 1072: 
 1073:             return maxEdge / minEdge;
 1074:         }

 1076:         /// <summary>
 1077:         /// Compares two quality configurations.
 1078:         /// Returns true if config1 is better than config2.
 1079:         /// Priority (in order):
 1080:         /// 1. Fewer topological inversions (CRITICAL)
 1081:         /// 2. Fewer element overlaps (IMPORTANT)
 1082:         /// 3. Lower inner triangle aspect ratio (QUALITY)
 1083:         /// 4. Lower outer triangle aspect ratio (SHAPE)
 1084:         /// </summary>
 1085:         private bool IsConfigurationBetter(
 1086:             (int inversions, int overlaps, double innerAspect, double outerAspect) config1,
 1087:             (int inversions, int overlaps, double innerAspect, double outerAspect) config2)
 1088:         {
 1089:             // Priority 1: Topological inversions are critical
 1090:             if (config1.inversions != config2.inversions)
 1091:                 return config1.inversions < config2.inversions;
 1092: 
 1093:             // Priority 2: Element overlaps are important
 1094:             if (config1.overlaps != config2.overlaps)
 1095:                 return config1.overlaps < config2.overlaps;
 1096: 
 1097:             // Priority 3: Inner triangle aspect ratio (lower is better)
 1098:             if (Math.Abs(config1.innerAspect - config2.innerAspect) > 0.01)
 1099:                 return config1.innerAspect < config2.innerAspect;
 1100: 
 1101:             // Priority 4: Outer triangle aspect ratio (lower is better)
 1102:             return config1.outerAspect < config2.outerAspect;
 1103:         }

  258:         public static Point2D FiberSurfaceArcPoint(Point2D fiberCenter, Point2D surfaceNode1,
  259:             Point2D surfaceNode2, double fiberRadius, double t)
  260:         {
  261:             double theta1 = Math.Atan2(surfaceNode1.Y - fiberCenter.Y,
  262:                 surfaceNode1.X - fiberCenter.X);
  263: 
  264:             double theta2 = Math.Atan2(surfaceNode2.Y - fiberCenter.Y,
  265:                 surfaceNode2.X - fiberCenter.X);
  266: 
  267:             double dTheta = theta2 - theta1;
  268: 
  269:             if (dTheta > Math.PI)
  270:                 dTheta -= 2.0 * Math.PI;
  271:             else if (dTheta < -Math.PI)
  272:                 dTheta += 2.0 * Math.PI;
  273: 
  274:             double theta = theta1 + t * dTheta;
  275: 
  276:             return new Point2D(
  277:                 fiberCenter.X + fiberRadius * Math.Cos(theta),
  278:                 fiberCenter.Y + fiberRadius * Math.Sin(theta));
  279:         }

  281:         public static Point2D PointAlong(Point2D a, Point2D b, double t)
  282:         {
  283:             return new Point2D(
  284:                 a.X + t * (b.X - a.X),
  285:                 a.Y + t * (b.Y - a.Y));
  286:         }

  288:         public static Point2D BilinearPoint(Point2D c0, Point2D c1, Point2D c2, Point2D c3, double xi, double eta)
  289:         {
  290:             double x =
  291:                 (1.0 - xi) * (1.0 - eta) * c0.X +
  292:                 xi * (1.0 - eta) * c1.X +
  293:                 xi * eta * c2.X +
  294:                 (1.0 - xi) * eta * c3.X;
  295: 
  296:             double y =
  297:                 (1.0 - xi) * (1.0 - eta) * c0.Y +
  298:                 xi * (1.0 - eta) * c1.Y +
  299:                 xi * eta * c2.Y +
  300:                 (1.0 - xi) * eta * c3.Y;
  301: 
  302:             return new Point2D(x, y);
  303:         }

  223:                 if (x > tolerance) return (1, 0);   // Right
  224:                 if (x < -tolerance) return (-1, 0); // Left
  225:             }
  226:             else if (Math.Abs(x) < tolerance) // Vertical projection
  227:             {
  228:                 if (y > tolerance) return (0, 1);   // Top
  229:                 if (y < -tolerance) return (0, -1); // Bottom
  230:             }
  231: 
  232:             return (0, 0);
  233:         }

  235:         /// <summary>
  236:         /// Checks if a shared edge is in counter-clockwise order within a triangle.
  237:         /// Based on MATLAB Triad.CheckIfSharedEdgeIsCCWOrder (lines 119-130).
  238:         /// Handles both fiber-fiber edges and fiber-boundary edges.
  239:         /// </summary>
  240:         private bool CheckIfSharedEdgeIsCCWOrder(Node[] triangleNodes, Node[] sharedEdgeNodes)
  241:         {
  242:             // Find indices of the shared edge nodes in the triangle
  243:             int idx1 = -1;
  244:             int idx2 = -1;
  245: 
  246:             for (int i = 0; i < triangleNodes.Length; i++)
  247:             {
  248:                 if (NodesMatch(triangleNodes[i], sharedEdgeNodes[0]))
  249:                     idx1 = i;
  250:                 if (NodesMatch(triangleNodes[i], sharedEdgeNodes[1]))
  251:                     idx2 = i;
  252:             }
  253: 
  254:             if (idx1 == -1 || idx2 == -1)
  255:                 throw new InvalidOperationException("Shared edge nodes not found in triangle");
  256: 
  257:             // Check if edge is in CCW order
  258:             if (idx1 == 0 && idx2 == 2)
  259:                 return false;
  260:             else if (idx1 < idx2 || (idx1 == triangleNodes.Length - 1 && idx2 == 0))
  261:                 return true;
  262:             else
  263:                 return false;
  264:         }

  266:         /// <summary>
  267:         /// Adjusts the middle nodes of fiber elements if fibers are too close together.
  268:         /// Based on MATLAB FE_Mesh.ChangeMiddleNodeIfFibersAreTooClose (lines 967-999).
  269:         /// </summary>
  270:         private (Point2D[] fiber1Nodes, Point2D[] fiber2Nodes) ChangeMiddleNodeIfFibersAreTooClose(
  271:             Point2D[] fiber1Nodes, Point2D[] fiber2Nodes, double radius1, double radius2)
  272:         {
  273:             double distanceBetweenFibers = MathHelper.CalcDistanceBetweenTwoPoints(fiber1Nodes[0], fiber2Nodes[0]);
  274:             double sumOfRadii = radius1 + radius2;
  275:             double ratio = sumOfRadii / distanceBetweenFibers;
  276: 
  277:             if (ratio >= 0.90)
  278:             {
  279:                 // Create vectors connecting fiber centers
  280:                 var v12 = MathHelper.MakeVector2D(fiber1Nodes[0], fiber2Nodes[0]);
  281:                 var v21 = MathHelper.MakeVector2D(fiber2Nodes[0], fiber1Nodes[0]);
  282: 
  283:                 // Calculate angles
  284:                 double t12 = Math.Atan2(v12.Y, v12.X);
  285:                 double t21 = Math.Atan2(v21.Y, v21.X);
  286: 
  287:                 // Update middle node locations
  288:                 var fiber1MiddleNode = new Point2D(
  289:                     fiber1Nodes[0].X + radius1 * Math.Cos(t12),
  290:                     fiber1Nodes[0].Y + radius1 * Math.Sin(t12));
  291: 
  292:                 var fiber2MiddleNode = new Point2D(
  293:                     fiber2Nodes[0].X + radius2 * Math.Cos(t21),
  294:                     fiber2Nodes[0].Y + radius2 * Math.Sin(t21));
  295: 
  296:                 // Copy arrays and update middle node (node 3 for 6-node element)
  297:                 var newFiber1Nodes = (Point2D[])fiber1Nodes.Clone();
  298:                 var newFiber2Nodes = (Point2D[])fiber2Nodes.Clone();
  299: 
  300:                 newFiber1Nodes[3] = fiber1MiddleNode;
  301:                 newFiber2Nodes[3] = fiber2MiddleNode;
  302: 
  303:                 return (newFiber1Nodes, newFiber2Nodes);
  304:             }
  305: 
  306:             return (fiber1Nodes, fiber2Nodes);
  307:         }

  309:         /// <summary>
  310:         /// Checks if a triangle shares an edge with two given nodes.
  311:         /// Matches both FiberId AND Offset for fiber nodes, or position for boundary nodes.
  312:         /// This prevents connecting original fibers to projected ones.
  313:         /// </summary>
  314:         private bool SharesEdge(Node[] triangleNodes, Node edgeNode1, Node edgeNode2)
  315:         {
  316:             int matchCount = 0;
  317: 
  318:             foreach (var node in triangleNodes)
  319:             {
  320:                 // Check if node matches edgeNode1
  321:                 if (NodesMatch(node, edgeNode1))
  322:                 {
  323:                     matchCount++;
  324:                 }
  325:                 // Check if node matches edgeNode2
  326:                 else if (NodesMatch(node, edgeNode2))
  327:                 {
  328:                     matchCount++;
  329:                 }
  330:             }
  331: 
  332:             return matchCount == 2;
  333:         }

  335:         /// <summary>
  336:         /// Checks if two nodes represent the same point.
  337:         /// For fiber nodes: matches FiberId and Offset.
  338:         /// For boundary nodes: matches position.
  339:         /// </summary>
  340:         private bool NodesMatch(Node node1, Node node2)
  341:         {
  342:             // For fiber nodes: match by FiberId and Offset
  343:             if (node1.FiberId.HasValue && node2.FiberId.HasValue)
  344:             {
  345:                 return node1.FiberId == node2.FiberId && node1.Offset == node2.Offset;
  346:             }
  347:             // For boundary nodes: match by position
  348:             else if (!node1.FiberId.HasValue && !node2.FiberId.HasValue)
  349:             {
  350:                 return Math.Abs(node1.P.X - node2.P.X) < NodeTolerance &&
  351:                        Math.Abs(node1.P.Y - node2.P.Y) < NodeTolerance &&
  352:                        node1.Offset == node2.Offset;
  353:             }
  354:             // Mixed fiber/boundary: no match
  355:             else
  356:             {
  357:                 return false;
  358:             }
  359:         }

  361:         /// <summary>
  362:         /// Creates a unique key for an edge including offset information.
  363:         /// Handles both fiber nodes (with FiberId) and boundary nodes (without FiberId).
  364:         /// This prevents matching original fibers with their projections.
  365:         /// </summary>
  366:         private string GetEdgeKey(Node node1, Node node2)
  367:         {
  368:             // Create a unique key that handles both fiber and boundary nodes
  369:             string key1 = node1.FiberId.HasValue
  370:                 ? $"F{node1.FiberId.Value}_{node1.Offset.ox}_{node1.Offset.oy}"
  371:                 : $"B{node1.P.X:F10}_{node1.P.Y:F10}_{node1.Offset.ox}_{node1.Offset.oy}";
  372: 
  373:             string key2 = node2.FiberId.HasValue
  374:                 ? $"F{node2.FiberId.Value}_{node2.Offset.ox}_{node2.Offset.oy}"
  375:                 : $"B{node2.P.X:F10}_{node2.P.Y:F10}_{node2.Offset.ox}_{node2.Offset.oy}";
  376: 
  377:             // Order-independent key
  378:             if (string.CompareOrdinal(key1, key2) < 0)
  379:                 return $"{key1}|{key2}";
  380:             else
  381:                 return $"{key2}|{key1}";
  382:         }

  481:         private bool TryAddPeriodicPair( int nodeIndex, Point2D projectedPoint, Dictionary<string, List<int>> nodeLookup,
  482:             List<(int, int)> pairs, HashSet<string> pairKeys, double tolerance)
  483:         {
  484:             int? matchingNodeIndex = FindGlobalNodeIndex(
  485:                 projectedPoint,
  486:                 nodeLookup,
  487:                 tolerance);
  488: 
  489:             if (!matchingNodeIndex.HasValue)
  490:                 return false;
  491: 
  492:             int i = nodeIndex;
  493:             int j = matchingNodeIndex.Value;
  494: 
  495:             if (i == j)
  496:                 return false;
  497: 
  498:             string pairKey = $"{Math.Min(i, j)}_{Math.Max(i, j)}";
  499: 
  500:             if (!pairKeys.Add(pairKey))
  501:                 return false;
  502: 
  503:             pairs.Add((i, j));
  504:             return true;
  505:         }
  506:         private int? FindGlobalNodeIndex( Point2D target, Dictionary<string, List<int>> nodeLookup, double tolerance)
  507:         {
  508:             string key = GetRoundedPointKey(target, tolerance);
  509: 
  510:             if (!nodeLookup.TryGetValue(key, out var candidateIndices))
  511:                 return null;
  512: 
  513:             foreach (int candidateIndex in candidateIndices)
  514:             {
  515:                 var candidate = _globalNodes[candidateIndex];
  516: 
  517:                 double dx = candidate.X - target.X;
  518:                 double dy = candidate.Y - target.Y;
  519: 
  520:                 if (Math.Sqrt(dx * dx + dy * dy) < tolerance)
  521:                     return candidateIndex;
  522:             }
  523: 
  524:             return null;
  525:         }

  600:         /// <summary>
  601:         /// Detects if any fibers in the triangle are too close to the opposite edge.
  602:         /// Uses MATLAB's two-tier threshold approach:
  603:         /// - Critical overlap (factor 20): fiber surface crossing edge - should throw
  604:         /// - Adjustment needed (factor 2): narrow inner triangle - should adjust midpoint
  605:         /// Based on MATLAB Triad.DetermineIfFibersOverlapTriad and willInteriorElementHaveToBeAdjusted.
  606:         /// </summary>
  607:         private FiberOverlapInfo[] DetectFiberOverlaps(Node[] triangleNodes, IReadOnlyList<Fiber> fibers)
  608:         {
  609:             var overlapInfo = new FiberOverlapInfo[3];
  610: 
  611:             for (int i = 0; i < 3; i++)
  612:             {
  613:                 overlapInfo[i] = new FiberOverlapInfo();
  614: 
  615:                 var currentNode = triangleNodes[i];
  616:                 if (currentNode.Type != NodeType.FiberCenter && currentNode.Type != NodeType.ProjectedFiber)
  617:                     continue;
  618: 
  619:                 var fiber = fibers[currentNode.FiberId.Value];
  620:                 var otherIndices = GetOtherIndices(i);
  621:                 var otherNode1 = triangleNodes[otherIndices[0]];
  622:                 var otherNode2 = triangleNodes[otherIndices[1]];
  623: 
  624:                 // Calculate minimum distance from fiber to the opposite edge
  625:                 double minDist = CalculatePointToLineDistance(
  626:                     currentNode.P,
  627:                     otherNode1.P,
  628:                     otherNode2.P);
  629: 
  630:                 // MATLAB threshold 1: Critical overlap (factor 20)
  631:                 // fiber.Radius + fiber.Radius/20 means surface is very close to edge
  632:                 double criticalThreshold = fiber.Radius + fiber.Radius / 30.0;
  633:                 overlapInfo[i].HasCriticalOverlap = minDist <= criticalThreshold;
  634: 
  635:                 // MATLAB threshold 2: Adjustment needed (factor 2)
  636:                 // fiber.Radius + fiber.Radius/2 means inner triangle is narrow
  637:                 double adjustmentThreshold = fiber.Radius + fiber.Radius / 2.0;
  638:                 overlapInfo[i].NeedsAdjustment = minDist <= adjustmentThreshold;
  639:             }
  640: 
  641:             return overlapInfo;
  642:         }

  644:         /// <summary>
  645:         /// Finds all shared edges between adjacent triangles and reconstructs their element nodes.
  646:         /// Returns edge data classified by edge type (TwoFibers, OneFiberOneBoundary, TwoBoundaries).
  647:         /// </summary>
  648:         private List<EdgeData> FindSharedEdgesForFiberElements(TriangulationMesh2D triangulation,
  649:             IReadOnlyList<Fiber> fibers, DebugOptions? dOptions = null)
  650:         {
  651:             var edgeDataList = new List<EdgeData>();
  652: 
  653:             // Store which triangle elements we've already built (mapping from triangle index to built interior triangle nodes)
  654:             var triangleElements = new Dictionary<int, Point2D[]>();
  655: 
  656:             // Reconstruct interior triangle element nodes for all triangles
  657:             for (int i = 0; i < triangulation.Triangles.Count; i++)
  658:             {
  659:                 var tri = triangulation.Triangles[i];
  660:                 var nodeA = triangulation.Nodes[tri[0]];
  661:                 var nodeB = triangulation.Nodes[tri[1]];
  662:                 var nodeC = triangulation.Nodes[tri[2]];
  663:                 var nodes = new[] { nodeA, nodeB, nodeC };
  664: 
  665:                 // Reconstruct the interior triangle element nodes the same way we built them
  666:                 var elementNodes = new Point2D[3];
  667:                 var overlapInfo = DetectFiberOverlaps(nodes, fibers);
  668: 
  669:                 // Check for critical overlaps and write error file if found
  670:                 bool hasCriticalOverlap = false;
  671:                 for (int j = 0; j < 3; j++)
  672:                 {
  673:                     if (overlapInfo[j].HasCriticalOverlap)
  674:                     {
  675:                         hasCriticalOverlap = true;
  676:                         break;
  677:                     }
  678:                 }
  679: 
  680:                 if (hasCriticalOverlap)
  681:                 {
  682:                     WriteCriticalOverlapErrorFile(nodes, fibers, overlapInfo, dOptions);
  683:                     // Skip processing this triangle since it has critical overlap
  684:                     continue;
  685:                 }
  686: 
  687:                 for (int j = 0; j < 3; j++)
  688:                 {
  689:                     var currentNode = nodes[j];
  690:                     var otherIndices = GetOtherIndices(j);
  691:                     var otherNode1 = nodes[otherIndices[0]];
  692:                     var otherNode2 = nodes[otherIndices[1]];
  693: 
  694:                     // Calculate surface point for fiber centers, use point as-is for boundary nodes
  695:                     if (currentNode.Type == NodeType.FiberCenter || currentNode.Type == NodeType.ProjectedFiber)
  696:                     {
  697:                         var fiber = fibers[currentNode.FiberId.Value];
  698:                         Point2D fiberCenter = currentNode.P;
  699: 
  700:                         elementNodes[j] = CalculateFiberSurfacePoint(
  701:                             fiberCenter,
  702:                             fiber.Radius,
  703:                             otherNode1.P,
  704:                             otherNode2.P,
  705:                             overlapInfo[j].NeedsAdjustment,
  706:                             j,
  707:                             overlapInfo);
  708:                     }
  709:                     else
  710:                     {
  711:                         // Boundary node - use as is
  712:                         elementNodes[j] = currentNode.P;
  713:                     }
  714:                 }
  715: 
  716:                 triangleElements[i] = elementNodes;
  717:             }
  718: 
  719:             // Find adjacent triangle pairs and classify edges
  720:             var processedEdges = new HashSet<string>();
  721: 
  722:             for (int i = 0; i < triangulation.Triangles.Count; i++)
  723:             {
  724:                 if (!triangleElements.ContainsKey(i))
  725:                     continue;
  726: 
  727:                 var tri1 = triangulation.Triangles[i];
  728:                 var nodes1 = new[] {
  729:                     triangulation.Nodes[tri1[0]],
  730:                     triangulation.Nodes[tri1[1]],
  731:                     triangulation.Nodes[tri1[2]]
  732:                 };
  733: 
  734:                 // Find all edges of this triangle
  735:                 var edges = new[] {
  736:                     (nodes1[0], nodes1[1]),
  737:                     (nodes1[1], nodes1[2]),
  738:                     (nodes1[2], nodes1[0])
  739:                 };
  740: 
  741:                 // Check each edge for adjacent triangles
  742:                 for (int edgeIdx = 0; edgeIdx < edges.Length; edgeIdx++)
  743:                 {
  744:                     var edge = edges[edgeIdx];
  745:                     var edgeKey = GetEdgeKey(edge.Item1, edge.Item2);
  746: 
  747:                     if (processedEdges.Contains(edgeKey))
  748:                         continue; // Already processed this edge
  749: 
  750:                     // Find adjacent triangle sharing this edge
  751:                     for (int j = i + 1; j < triangulation.Triangles.Count; j++)
  752:                     {
  753:                         if (!triangleElements.ContainsKey(j))
  754:                             continue;
  755: 
  756:                         var tri2 = triangulation.Triangles[j];
  757:                         var nodes2 = new[] {
  758:                             triangulation.Nodes[tri2[0]],
  759:                             triangulation.Nodes[tri2[1]],
  760:                             triangulation.Nodes[tri2[2]]
  761:                         };
  762: 
  763:                         // Check if this triangle shares the edge
  764:                         bool sharesEdge = SharesEdge(nodes2, edge.Item1, edge.Item2);
  765: 
  766:                         if (sharesEdge)
  767:                         {
  768:                             // Mark edge as processed
  769:                             processedEdges.Add(edgeKey);
  770: 
  771:                             // Classify the edge type based on the shared edge nodes
  772:                             var edgeType = ClassifyEdgeType(edge.Item1, edge.Item2);
  773: 
  774:                             // Store edge data with classification
  775:                             var edgeData = new EdgeData(
  776:                                 nodes1,
  777:                                 nodes2,
  778:                                 triangleElements[i],
  779:                                 triangleElements[j],
  780:                                 new[] { edge.Item1, edge.Item2 },
  781:                                 edgeType);
  782: 
  783:                             edgeDataList.Add(edgeData);
  784: 
  785:                             break; // Only one adjacent triangle per edge
  786:                         }
  787:                     }
  788:                 }
  789:             }
  790: 
  791:             return edgeDataList;
  792:         }

 1802:         /// <summary>
 1803:         /// Builds 1 triangular matrix element for a shared edge with one fiber and one boundary node.
 1804:         /// Creates a 6-node triangular element connecting the fiber surface to the boundary point.
 1805:         /// </summary>
 1806:         private void BuildTriangularMatrixElement(Node[] triangle1Nodes, Node[] triangle2Nodes,
 1807:             Point2D[] triangle1ElementNodes, Point2D[] triangle2ElementNodes, Node[] sharedEdgeNodes,
 1808:             IReadOnlyList<Fiber> fibers, ElementConfig config)
 1809:         {
 1810:             // Identify which node is the fiber and which is the boundary
 1811:             Node fiberNode, boundaryNode;
 1812:             Fiber fiber;
 1813: 
 1814:             if (sharedEdgeNodes[0].Type == NodeType.FiberCenter || sharedEdgeNodes[0].Type == NodeType.ProjectedFiber)
 1815:             {
 1816:                 fiberNode = sharedEdgeNodes[0];
 1817:                 boundaryNode = sharedEdgeNodes[1];
 1818:                 fiber = fibers[fiberNode.FiberId.Value];
 1819:             }
 1820:             else
 1821:             {
 1822:                 fiberNode = sharedEdgeNodes[1];
 1823:                 boundaryNode = sharedEdgeNodes[0];
 1824:                 fiber = fibers[fiberNode.FiberId.Value];
 1825:             }
 1826: 
 1827:             // Get positions
 1828:             var fiberCenter = fiberNode.P;
 1829:             var boundaryPoint = boundaryNode.P;
 1830: 
 1831:             // Find element nodes by matching the nodes in both triangles
 1832:             var fiberNode_Tri1 = FindInteriorTriangleNodeByNode(triangle1Nodes, triangle1ElementNodes, fiberNode);
 1833:             var fiberNode_Tri2 = FindInteriorTriangleNodeByNode(triangle2Nodes, triangle2ElementNodes, fiberNode);
 1834:             var boundaryNode_Tri1 = FindInteriorTriangleNodeByNode(triangle1Nodes, triangle1ElementNodes, boundaryNode);
 1835:             var boundaryNode_Tri2 = FindInteriorTriangleNodeByNode(triangle2Nodes, triangle2ElementNodes, boundaryNode);
 1836:

 1838:             if (Math.Abs(boundaryNode_Tri1.X - boundaryNode_Tri2.X) > NodeTolerance ||

 1840:             {
 1841:                 throw new InvalidOperationException("Boundary nodes from adjacent triangles don't match");
 1842:             }

 1844:             // Determine if shared edge is in CCW order in triangle 1
 1845:             bool isEdgeCCW = CheckIfSharedEdgeIsCCWOrder(triangle1Nodes, sharedEdgeNodes);
 1846: 
 1847:             // Determine fiber node order (creates curved fiber surface with 6 nodes)
 1848:             var fiberNodes = DetermineFiberNodeOrder(fiberCenter, fiberNode_Tri1, fiberNode_Tri2, fiber.Radius, isEdgeCCW);
 1849: 
 1850:             // Build 6-node triangular matrix element
 1851:             // Node layout for 6-node triangle:
 1852:             //     node2
 1853:             //      /\
 1854:             //     /  \
 1855:             // n5 /    \ n3
 1856:             //   /      \
 1857:             //  /________\
 1858:             // node0  n4  node1

 1872:                 triangleNodes[2] = boundaryNode_Tri1; // Boundary point
 1873:             }
 1874:             else
 1875:             {
 1876:                 // Boundary node is first in shared edge
 1877:                 triangleNodes[0] = boundaryNode_Tri1; // Boundary point

 1879:                 triangleNodes[2] = fiberNodes[4];  // Second point on fiber surface

 1993:         /// <summary>
 1994:         /// Calculates the minimum distance from a point to a line segment.
 1995:         /// </summary>
 1996:         private double CalculatePointToLineDistance(Point2D point, Point2D lineStart, Point2D lineEnd)
 1997:         {
 1998:             double dx = lineEnd.X - lineStart.X;
 1999:             double dy = lineEnd.Y - lineStart.Y;
 2000: 
 2001:             if (Math.Abs(dx) < 1e-10 && Math.Abs(dy) < 1e-10)
 2002:             {
 2003:                 // Line is actually a point
 2004:                 return MathHelper.CalcDistanceBetweenTwoPoints(point, lineStart);
 2005:             }
 2006: 
 2007:             // Calculate distance using cross product formula
 2008:             double numerator = Math.Abs(dy * point.X - dx * point.Y + lineEnd.X * lineStart.Y - lineEnd.Y * lineStart.X);
 2009:             double denominator = Math.Sqrt(dx * dx + dy * dy);
 2010: 
 2011:             return numerator / denominator;
 2012:         }

 2014:         /// <summary>
 2015:         /// Calculates a point on the surface of a fiber that faces the interior of the triangle.
 2016:         /// Uses vector addition to find the bisector - no angle calculations needed.
 2017:         /// Handles overlap cases by using edge directions instead of bisector.
 2018:         /// </summary>
 2019:         private Point2D CalculateFiberSurfacePoint(Point2D fiberCenter, double fiberRadius, Point2D otherPoint1,
 2020:             Point2D otherPoint2, bool needsAdjustment, int currentIndex, FiberOverlapInfo[] allOverlaps)
 2021:         {
 2022:             // Create vectors from fiber center to the other two points
 2023:             var vec1 = MathHelper.MakeVector2D(fiberCenter, otherPoint1);
 2024:             var vec2 = MathHelper.MakeVector2D(fiberCenter, otherPoint2);
 2025: 
 2026:             // Normalize both vectors
 2027:             var vec1Normalized = Normalize(vec1);
 2028:             var vec2Normalized = Normalize(vec2);
 2029: 
 2030:             Point2D direction;
 2031: 
 2032:             // MATLAB logic: If ANY fiber in the triangle needs adjustment (narrow gap),
 2033:             // we adjust the OTHER fibers (not the one with the narrow gap).
 2034:             // This is the key insight from line 74 of Triad.m:
 2035:             //   shiftCenterDueToOverlap = ~obj.willInteriorElementHaveOverlapWithFiber(i)
 2036:             // The negation means: "shift if the CURRENT fiber is NOT the overlapping one"
 2037: 
 2038:             // First, check if any fiber needs adjustment
 2039:             int overlapIndex = -1;
 2040:             bool anyFiberNeedsAdjustment = false;
 2041:             for (int i = 0; i < allOverlaps.Length; i++)
 2042:             {
 2043:                 if (allOverlaps[i].NeedsAdjustment)
 2044:                 {
 2045:                     anyFiberNeedsAdjustment = true;
 2046:                     overlapIndex = i;
 2047:                     break;
 2048:                 }
 2049:             }
 2050: 
 2051:             // Now, if any fiber needs adjustment AND the current fiber is NOT that fiber,
 2052:             // then we should adjust the current fiber's surface point
 2053:             bool shouldAdjustCurrentFiber = anyFiberNeedsAdjustment && !needsAdjustment;
 2054: 
 2055:             if (shouldAdjustCurrentFiber)
 2056:             {
 2057:                 // Adjust this fiber's surface point away from the overlapping fiber
 2058:                 // This matches MATLAB's AdjustMidPointDueToOverlap
 2059:                 direction = GetEdgeDirectionForOverlap(vec1Normalized, vec2Normalized, currentIndex, overlapIndex);
 2060:             }
 2061:             else
 2062:             {
 2063:                 // Normal case: use bisector
 2064:                 var bisectorDirection = new Point2D(
 2065:                     vec1Normalized.X + vec2Normalized.X,
 2066:                     vec1Normalized.Y + vec2Normalized.Y);
 2067:                 direction = Normalize(bisectorDirection);
 2068:             }
 2069: 
 2070:             // Calculate point on fiber surface
 2071:             return new Point2D(
 2072:                 fiberCenter.X + fiberRadius * direction.X,
 2073:                 fiberCenter.Y + fiberRadius * direction.Y);
 2074:         }

 2089:         /// <summary>
 2090:         /// Determines which edge direction to use when overlap is detected.
 2091:         /// Based on MATLAB's AdjustMidPointDueToOverlap logic.
 2092:         /// 
 2093:         /// IMPORTANT: MATLAB has special handling for the "middle edge" (fiber index 1 in 0-based).
 2094:         /// For fiber 1, T_Unit is based on VAC instead of VAB, which effectively swaps vec1 and vec2.
 2095:         /// </summary>
 2096:         private Point2D GetEdgeDirectionForOverlap(Point2D vec1, Point2D vec2, int currentIdx, int overlapIdx)
 2097:         {
 2098:             // MATLAB uses isMiddleEdge flag: if(i == 2) in MATLAB 1-based means (i == 1) in 0-based
 2099:             // For the middle fiber, MATLAB swaps which vector is considered "first"
 2100:             bool isMiddleEdge = (currentIdx == 1);
 2101: 
 2102:             // When overlap is detected, use the vector pointing to one of the edges
 2103:             // rather than the bisector, to avoid the interior triangle overlapping the fiber
 2104:             //
 2105:             // This matches MATLAB's AdjustMidPointDueToOverlap logic:
 2106:             // - vec1 corresponds to T_Unit (first edge angle)
 2107:             // - vec2 corresponds to T_Unit + TAB_AC (second edge angle)
 2108:             // - BUT for middle edge (currentIdx == 1), these are swapped in MATLAB
 2109:             // - overlapIdx is 0-based (0, 1, 2) corresponding to MATLAB's 1-based (1, 2, 3)
 2110:             // - currentIdx is 0-based (0, 1, 2) corresponding to MATLAB's idx (1, 2, 3)
 2111: 
 2112:             // If this is the middle edge, swap vec1 and vec2 to match MATLAB's isMiddleEdge behavior
 2113:             if (isMiddleEdge)
 2114:             {
 2115:                 var temp = vec1;
 2116:                 vec1 = vec2;
 2117:                 vec2 = temp;
 2118:             }
 2119: 
 2120:             switch (overlapIdx)
 2121:             {
 2122:                 case 0:
 2123:                     // Fiber 0 overlaps (MATLAB case 1)
 2124:                     if (currentIdx == 1)
 2125:                         return vec1;  // MATLAB: idx==2 → T_Unit (but swapped for middle edge)
 2126:                     else if (currentIdx == 2)
 2127:                         return vec2;  // MATLAB: idx==3 → T_Unit + TAB_AC
 2128:                     break;
 2129:                 case 1:
 2130:                     // Fiber 1 overlaps (MATLAB case 2)
 2131:                     if (currentIdx == 0)
 2132:                         return vec2;  // MATLAB: idx==1 → T_Unit + TAB_AC
 2133:                     else if (currentIdx == 2)
 2134:                         return vec1;  // MATLAB: idx==3 → T_Unit
 2135:                     break;
 2136:                 case 2:
 2137:                     // Fiber 2 overlaps (MATLAB case 3)
 2138:                     if (currentIdx == 0)
 2139:                         return vec1;  // MATLAB: idx==1 → T_Unit
 2140:                     else if (currentIdx == 1)
 2141:                         return vec2;  // MATLAB: idx==2 → T_Unit + TAB_AC (but swapped for middle edge)
 2142:                     break;
 2143:             }
 2144: 
 2145:             // Default: use bisector (shouldn't reach here in normal operation)
 2146:             var bisector = new Point2D(vec1.X + vec2.X, vec1.Y + vec2.Y);
 2147:             return Normalize(bisector);
 2148:         }

 2150:         /// <summary>
 2151:         /// Normalizes a 2D vector to unit length.
 2152:         /// </summary>
 2153:         private Point2D Normalize(Point2D vector)
 2154:         {
 2155:             double length = Math.Sqrt(vector.X * vector.X + vector.Y * vector.Y);
 2156:             if (length < 1e-10)  // Avoid division by zero
 2157:             {
 2158:                 return new Point2D(1, 0);  // Default direction
 2159:             }
 2160:             return new Point2D(vector.X / length, vector.Y / length);
 2161:         }
